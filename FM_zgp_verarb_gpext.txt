FUNCTION /vwk/zgp_verarb_gpext.
*"----------------------------------------------------------------------
*"*"Lokale Schnittstelle:
*"  IMPORTING
*"     REFERENCE(IV_MODUS) TYPE  CHAR1
*"     REFERENCE(IV_DIALOG_ERLAUBT) TYPE  CHAR1
*"     REFERENCE(IV_TEST) TYPE  CHAR1 DEFAULT SPACE
*"  EXPORTING
*"     REFERENCE(EV_ZIEL_ID) TYPE  /VWK/ZGP_ZIEL_ID
*"     REFERENCE(ES_GPEXT_ORIG) TYPE  /VWK/ZGPS_GPEXT
*"  TABLES
*"      ET_MESSAGE STRUCTURE  BUS0MSG1
*"  CHANGING
*"     REFERENCE(CS_GPEXT) TYPE  /VWK/ZGPS_GPEXT
*"     REFERENCE(CS_GPINT) TYPE  /VWK/ZGPS_GPEXT
*"     REFERENCE(CS_GPEXT_KOMM) TYPE  /VWK/ZGP_GPEXT_KOMM
*"     REFERENCE(CS_DYNPRO) TYPE  /VWK/ZGP_GPEXT_DYN9600
*"     REFERENCE(CV_PARTNER) TYPE  BU_PARTNER
*"  EXCEPTIONS
*"      VERARBEITUNG_FEHLERHAFT
*"----------------------------------------------------------------------
* Lokale Datendeklarationen
  DATA: lv_gp_fehler      TYPE xfeld,
        lv_test           TYPE char1,
        lt_message        TYPE bus0msg1_t,
        lv_msgtype        TYPE symsgty,
*  BOI : 017
        lv_udate          TYPE sy-datum,
        lv_utime          TYPE sy-uzeit,
        lv_timestamp      TYPE timestamp,
        lv_partner        TYPE cdhdr-objectid,
        lt_but0bk         TYPE STANDARD TABLE OF vbut0bk,
        lt_cdtxt          TYPE STANDARD TABLE OF cdtxt,
*  EOI :017
        lv_register_type  TYPE string,                           "021++
        lo_zgp_dao_but000 TYPE REF TO /vwk/cl_zgp_dao_but000.   "021++


  CONSTANTS : lc_delete TYPE char14 VALUE 'DELETE'.  "016++
  CONSTANTS: lc_tzone_utc     TYPE tznzone VALUE 'UTC'. "017++

  IF cs_gpext-s-ziel_id NE gv_ziel_id  OR
     cs_gpext-s-partner NE cv_partner  AND
     cs_gpext-s-partner IS NOT INITIAL.
* Bei Wechsel der GP-Nummer oder der Ziel-ID: Neue Prüfung
    gv_ziel_id = cs_gpext-s-ziel_id.
    PERFORM entsperren_partner USING cv_partner.
    CLEAR: cv_partner,
           cs_gpext_komm-aenderung_abgefragt,
           cs_gpext_komm-mailadr_alt,
           cs_gpext_komm-addrnr_std,
           cs_gpext_komm-addrverw.
    REFRESH: cs_gpext_komm-addrnr_rest.
  ELSEIF   cs_gpext-s-ziel_id IS INITIAL     AND
           cs_gpext-s-partner IS INITIAL     AND
           cv_partner         IS NOT INITIAL AND
           cs_gpext_komm-mailadr_alt IS INITIAL.
    PERFORM entsperren_partner USING cv_partner.
    CLEAR: cv_partner,
*           cs_gpext_komm-aenderung_abgefragt, ++018
           cs_gpext_komm-addrnr_std,
           cs_gpext_komm-addrverw.
    REFRESH: cs_gpext_komm-addrnr_rest.
  ENDIF.

  "Prüfen, ob ein Compliance-Verdacht (CCF, KYC) vorliegt
  PERFORM pruefen_compliance_verdacht USING    iv_dialog_erlaubt
                                               cs_gpext
                                      CHANGING cs_gpext_komm.

  IF cs_gpext-s-fehlerart = gc_fehler_host_zuordnung.
*   Bestätigung der falschen Host-Zuordnung abfragen
    PERFORM bestaetigung_hostzuordnung  USING     iv_dialog_erlaubt
                                                  cs_gpext
                                        CHANGING  cs_gpext_komm.
  ENDIF.

* Automatische Feldanreicherung-/löschung
  PERFORM daten_vorbereitung            USING     iv_dialog_erlaubt
                                        CHANGING  cs_gpext_komm
                                                  cs_gpext
                                                  et_message[].

  "Korrigierte bzw. ergänzte Daten als neue Originaldaten zurückgeben
  es_gpext_orig = cs_gpext.

* Formalprüfungen (z.B. Bankverbindungsdaten)
  PERFORM formalpruefungen USING iv_dialog_erlaubt
                                 cs_gpext_komm
                        CHANGING cs_gpext
                                 lt_message.

  APPEND LINES OF lt_message[] TO et_message.
*BOI:019
  cs_gpext-g-el_kommunikation = abap_false.
*EOI:019

* BOI 020
* When input system is HR-DATEN and no register data is sent then this record is meant to be routed to ZAV
  IF cs_gpext-s-inputsystem = /vwk/if_zgp_consts_inp=>scu_inpsys_hr_daten.
*   Check if register data is sent via XML or Not, If not then register data is not available in both AKB and ZGP
*   Sending such record to ZAV for manual search of imp Data
    IF cs_gpext-g-z_zgp_comrg      IS INITIAL AND
       cs_gpext-g-z_zgp_comrg_dat  IS INITIAL AND
       cs_gpext-g-z_zgp_coprg      IS INITIAL AND
       cs_gpext-g-z_zgp_coprg_dat  IS INITIAL AND
       cs_gpext-g-z_zgp_clbrg      IS INITIAL AND
       cs_gpext-g-z_zgp_clbrg_dat  IS INITIAL AND
       cs_gpext-g-z_zgp_prtrg      IS INITIAL AND
       cs_gpext-g-z_zgp_prtrg_dat  IS INITIAL AND
       cs_gpext-g-z_zgp_infct      IS INITIAL AND
*          cs_gpext-g-bukrs = /vwk/if_zgp_consts=>scu_bukrs_kredis.       "++023

       " Raise an failure to route to ZAV and to stop further processing.
        gv_fehlerart = gc_fehler_gl_pflege.
      MESSAGE e148
      RAISING verarbeitung_fehlerhaft.
    ENDIF.
*     BOI 021
*     Now, since the register data exists let's make the plausibility check for the
*     Legal form of the BP VS Register prefix

*     First check which register field is filled
    IF cs_gpext-g-z_zgp_comrg IS NOT INITIAL.
      lv_register_type = cs_gpext-g-z_zgp_comrg.

    ELSEIF cs_gpext-g-z_zgp_coprg IS NOT INITIAL.
      lv_register_type = cs_gpext-g-z_zgp_coprg.

    ELSEIF cs_gpext-g-z_zgp_clbrg IS NOT INITIAL.
      lv_register_type = cs_gpext-g-z_zgp_clbrg.

    ELSEIF cs_gpext-g-z_zgp_prtrg IS NOT INITIAL.
      lv_register_type = cs_gpext-g-z_zgp_prtrg.
    ENDIF.
*     Now, derive register prefix from the above (reg. prefix + reg. Number) string e.g., HRA from "HRA 2343 24"
    IF lv_register_type IS NOT INITIAL.
      SPLIT lv_register_type AT space INTO lv_register_type DATA(dummy).  "dummy is leftover part, not useful
    ENDIF.
*     Now, get the Legal form of the BP
    IF cs_gpext-s-partner IS NOT INITIAL.
      TRY.
          CREATE OBJECT lo_zgp_dao_but000.
          DATA(lv_legal_form) = lo_zgp_dao_but000->read_legal_enty_by_id( iv_partner = cs_gpext-s-partner ).
        CATCH /vwk/cx_zgp_but000_no_data.
          " Do nothing, this doesn't happen mostly.
      ENDTRY.
    ENDIF.
*      Read plausible register type from the customizing for the given legal form
    SELECT SINGLE reg_check
           INTO @DATA(lv_expected_reg)
           FROM /vwk/zgp_leg_frm
           WHERE legal_form = @lv_legal_form.
*      Validate register type
    IF lv_expected_reg IS NOT INITIAL AND lv_register_type <> lv_expected_reg.
      " Raise an failure to route to ZAV and to stop further processing.
      gv_fehlerart = gc_fehler_gp_pflege.
      MESSAGE e149 WITH lv_register_type lv_legal_form RAISING verarbeitung_fehlerhaft.
    ENDIF.
*    EOI 021
  ENDIF.
* Checking if table steurnummern is already filled with tax number of type 'DE0' If already filled,
* then removing the tax number and tax type from gpext to avoid failure of input record and
* update the table international indentification numbers /vwk/zgp_intidno
  IF cs_gpext-s-inputsystem = /vwk/if_zgp_consts_inp=>scu_inpsys_ust_id_akb.
*   For roles other than ZGP001 no filling of steuernummern table since fields not there in GUI
    IF cs_gpext-s-rltyp <> /vwk/if_zgp_consts=>scu_role_zgp.
      PERFORM filling_zgp_intidno USING cs_gpext.
      CLEAR cs_gpext-t.
    ELSE.
      DATA(rv_exists) = abap_false.
      IF cs_gpext-s-partner IS NOT INITIAL.
        PERFORM check_steueurnummern_exists USING     cs_gpext-s-partner
                                            CHANGING  rv_exists.
        IF rv_exists = abap_true AND iv_test IS INITIAL.
*     Updating table /vwk/zgp_intidno and clearing stored gpext tax info to avoid failure of input record since
*     DE0 type taxnum already exists and duplicate is not allowed.
          PERFORM filling_zgp_intidno USING cs_gpext.
          CLEAR cs_gpext-t.
        ELSE.
*          Do nothing, continue processing input record.
        ENDIF.
      ENDIF.
    ENDIF.
  ENDIF.
* EOI 020
* Prüfung und Neuanlage/Änderung Geschäftspartner

  PERFORM geschaeftspartner  TABLES   et_message
                             USING    iv_modus
                                      iv_test
                                      iv_dialog_erlaubt
                             CHANGING cs_gpext_komm
                                      cs_gpext
                                      cs_gpint
                                      cv_partner
                                      ev_ziel_id
                                      cs_dynpro
                                      lv_gp_fehler.

  " P-Tabelle(MEV) nochmals zuweisen, da sie ggf. mit Vorbelegungen ergänzt wird
  es_gpext_orig-p = cs_gpext-p.
* Modus übernehmen
  lv_test = iv_test.

  IF iv_test IS INITIAL AND NOT lv_gp_fehler IS INITIAL.
*   Echtlauf; GP-Fehler vorhanden:
*   - Umschalten auf Testmodus, um auch die Vertragsdaten zu prüfen
    lv_test = gc_kreuz.
  ENDIF.

  IF NOT lv_gp_fehler IS INITIAL.
    MESSAGE e023
*     Die Geschäftspartnerdaten sind fehlerfrei
    RAISING verarbeitung_fehlerhaft.
  ENDIF.

  IF iv_test = space AND
     cs_gpext_komm-inpsys-altdaten = gc_kreuz AND
     NOT cs_gpext-s-dest_system_name IS INITIAL AND
     NOT cs_gpext-s-ziel_id IS INITIAL.
*   Auskünfte der neuen ZGP-Nummer zuordnen --> Nur Echtlauf Vorlage
    PERFORM auskunft_zuordnen USING cs_gpext
                                    cv_partner.
  ENDIF.


*BOI : 016
  IF iv_test = space AND cv_partner IS NOT INITIAL.
    IF cs_gpext-b IS NOT INITIAL.
      LOOP AT cs_gpext-b ASSIGNING FIELD-SYMBOL(<lfs_gpext_b>) WHERE method = lc_delete.
        IF <lfs_gpext_b>-iban IS NOT INITIAL.
          "BOI 024
          "Commented 024
*          SELECT  SINGLE * INTO  @DATA(ls_tban) FROM tiban WHERE iban = @<lfs_gpext_b>-iban.
          SELECT * INTO  @DATA(ls_tban) FROM tiban UP TO 1 ROWS
             WHERE iban = @<lfs_gpext_b>-iban
            ORDER BY PRIMARY KEY.
          ENDSELECT.
          IF ls_tban IS NOT INITIAL.
*BOI : 017
            "Commented 024
*            SELECT SINGLE * FROM but0bk INTO @DATA(ls_but0bk) WHERE partner = @cv_partner AND banks = @ls_tban-banks AND
*                                     bankl = @ls_tban-bankl AND bankn = @ls_tban-bankn.
            SELECT * FROM but0bk UP TO 1 ROWS
              INTO @DATA(ls_but0bk) WHERE partner = @cv_partner AND banks = @ls_tban-banks AND
                         bankl = @ls_tban-bankl AND bankn = @ls_tban-bankn
              ORDER BY PRIMARY KEY.
            ENDSELECT.
            "EOI 024
            IF sy-subrc IS INITIAL.
*lock business partner
              CALL FUNCTION 'BUP_ENQUEUE'
                EXPORTING
                  i_partner    = cv_partner
                  i_aktyp      = '02'
                  i_aktdb      = '02'
                  i_xsave      = abap_true
                EXCEPTIONS
                  foreign_lock = 1
                  OTHERS       = 2.

              CASE sy-subrc.
                WHEN 1.
                  MESSAGE ID sy-msgid TYPE 'E' NUMBER sy-msgno
                          WITH sy-msgv1 sy-msgv2 sy-msgv3 sy-msgv4.
                WHEN 2.
                  MESSAGE x066(/vwk/zgp) WITH 'BUP_ENQUEUE'.
              ENDCASE.
              DELETE FROM but0bk WHERE partner = cv_partner AND banks = ls_tban-banks AND
                                       bankl = ls_tban-bankl AND bankn = ls_tban-bankn.
              IF sy-subrc IS INITIAL.
                ls_but0bk-iban = <lfs_gpext_b>-iban.
                APPEND ls_but0bk TO lt_but0bk.
*update change history
                lv_partner = cv_partner.
                GET TIME STAMP FIELD lv_timestamp.
                CONVERT TIME STAMP lv_timestamp TIME ZONE lc_tzone_utc
                  INTO DATE lv_udate TIME lv_utime.
                CALL FUNCTION 'BUPA_BUP_WRITE_DOCUMENT'
                  EXPORTING
                    objectid                = lv_partner
                    tcode                   = sy-tcode
                    utime                   = lv_utime
                    udate                   = lv_udate
                    username                = sy-uname
                    object_change_indicator = 'D'
                    upd_but0bk              = 'D'
                  TABLES
                    ybut0bk                 = lt_but0bk
                    icdtxt_bupa_bup         = lt_cdtxt.
              ENDIF.
** Unlock business partner
              CALL FUNCTION 'BUP_DEQUEUE'
                EXPORTING
                  i_partner = cv_partner
                  i_aktyp   = '02'
                  i_aktdb   = '02'
                  i_xsave   = abap_true.
            ENDIF.
          ENDIF.
        ENDIF.
      ENDLOOP.
    ENDIF.
    CLEAR: lt_but0bk, ls_but0bk, ls_tban, lv_partner,lv_utime,lv_udate.
*EOI : 017
  ENDIF.
*EOI : 016
* BOI 020
* Updating table /vwk/zgp_intidno now, since the input record is processed successfully and have updated the
* Table Steuernummern with the tax type DE0.
  IF ( iv_test IS INITIAL ) AND
    ( cv_partner IS NOT INITIAL ) AND
    ( cs_gpext-s-inputsystem = /vwk/if_zgp_consts_inp=>scu_inpsys_ust_id_akb ).
    PERFORM filling_zgp_intidno USING cs_gpext.
  ENDIF.
* EOI 020
  IF iv_test IS INITIAL AND NOT cv_partner IS INITIAL.
*   Vorbereitung und Durchführung der Verteilerlistenneuanlage
*   für Daten aus dem S-SATZ und den D-Sätzen
*   (nur wenn keine GP-Prüfung und GP-Nummer gefüllt)
    PERFORM verteilerliste  USING     cv_partner
                                      iv_test
                                      iv_dialog_erlaubt
                            CHANGING  cs_gpext_komm
                                      cs_gpext
                                      ev_ziel_id.
  ENDIF.

ENDFUNCTION.
